describe("field_comparison rules", () => {
  const dateRange: ClientDefinition = {
    client: "test",
    record_type: "date_range",
    fields: [
      { name: "start_date", label: "Start date", type: "date", required: true },
      { name: "end_date", label: "End date", type: "date", required: true },
    ],
    rules: [
      { type: "field_comparison", field: "end_date", operator: "gte", compare_to: "start_date" },
    ],
  };

  const optionalDateRange: ClientDefinition = {
    ...dateRange,
    fields: dateRange.fields.map((f) => ({ ...f, required: false })),
  };

  function comparisonDef(operator: ComparisonOperator): ClientDefinition {
    return {
      client: "test",
      record_type: "comparison",
      fields: [
        { name: "a", label: "A", type: "number", required: false },
        { name: "b", label: "B", type: "number", required: false },
      ],
      rules: [{ type: "field_comparison", field: "a", operator, compare_to: "b" }],
    };
  }

  it("passes when the comparison holds, including equal dates for gte", () => {
    expect(
      validate(dateRange, { start_date: "2026-01-01", end_date: "2026-01-01" })
    ).toEqual([]);
    expect(
      validate(dateRange, { start_date: "2026-01-01", end_date: "2026-01-05" })
    ).toEqual([]);
  });

  it("fails when the comparison is violated, reporting against the subject field", () => {
    const result = validate(dateRange, {
      start_date: "2026-09-10",
      end_date: "2026-09-01",
    });
    expect(result).toEqual([
      { field: "end_date", error: "Must be on or after Start date" },
    ]);
  });

  it("supports lt", () => {
    expect(validate(comparisonDef("lt"), { a: 1, b: 2 })).toEqual([]);
    expect(errorsFor(validate(comparisonDef("lt"), { a: 2, b: 2 }), "a")).toHaveLength(1);
  });

  it("supports lte", () => {
    expect(validate(comparisonDef("lte"), { a: 2, b: 2 })).toEqual([]);
    expect(errorsFor(validate(comparisonDef("lte"), { a: 3, b: 2 }), "a")).toHaveLength(1);
  });

  it("supports gt", () => {
    expect(validate(comparisonDef("gt"), { a: 3, b: 2 })).toEqual([]);
    expect(errorsFor(validate(comparisonDef("gt"), { a: 2, b: 2 }), "a")).toHaveLength(1);
  });

  it("supports gte", () => {
    expect(validate(comparisonDef("gte"), { a: 2, b: 2 })).toEqual([]);
    expect(errorsFor(validate(comparisonDef("gte"), { a: 1, b: 2 }), "a")).toHaveLength(1);
  });

  it("supports eq", () => {
    expect(validate(comparisonDef("eq"), { a: 2, b: 2 })).toEqual([]);
    expect(errorsFor(validate(comparisonDef("eq"), { a: 1, b: 2 }), "a")).toHaveLength(1);
  });

  it("supports neq", () => {
    expect(validate(comparisonDef("neq"), { a: 1, b: 2 })).toEqual([]);
    expect(errorsFor(validate(comparisonDef("neq"), { a: 2, b: 2 }), "a")).toHaveLength(1);
  });

  it("skips the rule when the dependency is missing, producing no error", () => {
    expect(
      validate(optionalDateRange, { end_date: "2026-01-01" })
    ).toEqual([]);
  });

  it("skips the rule when the dependency is missing and required, reporting only the base error", () => {
    const result = validate(dateRange, { end_date: "2026-01-01" });
    expect(result).toEqual([{ field: "start_date", error: "This field is required" }]);
  });

  it("skips the rule when the dependency is invalid, reporting only the base error", () => {
    const result = validate(dateRange, {
      start_date: "not-a-date",
      end_date: "2026-01-01",
    });
    expect(result).toEqual([
      { field: "start_date", error: "Must be a date in YYYY-MM-DD format" },
    ]);
  });
});

Part 4

Six Decisions

Context

The library you just extended is the seed of a multi-tenant platform: one codebase serving many client organisations, where every client difference is configuration rather than code. Answer with that platform in mind. Decisions 1 and 2 refer directly to the code you wrote in Part 3.

  1. 01
    Database — I’d choose PostgreSQL. JSONB lets each client have different fields without changing the database structure for every client. PostgreSQL also gives strong transactions, relationships, and tenant isolation through RLS. The tradeoff is added work managing JSONB indexes and RLS policies. If we used MongoDB, my Part 3 validator would stay the same; I would only change how data is stored, retrieved, and isolated between clients.

    427 / 600 characters

  2. 02
    Scope — I deliberately did not add cross-record rules, such as preventing double-booking or duplicate values. My validator only checks one record at a time and has no database access. Adding these rules would introduce storage and concurrency concerns into a simple, client-agnostic validator. I stopped there because cross-record constraints should be enforced at the database layer, where they can be guaranteed safely.

    413 / 600 characters

  3. 03
    Isolation — I'd choose a shared database with strict tenant isolation using tenant_id and RLS. It keeps infrastructure cost low, onboarding simple, and lets one platform serve many clients efficiently. The tradeoff is that logical isolation cannot satisfy a client requiring their data to never share infrastructure with others. I accept losing that client rather than making every tenant pay the cost and operational complexity of dedicated infrastructure.

    445 / 600 characters

  4. 04
    Leadership — I explain the bug, then require the developer to fix it and add a regression test themselves. I don't take over the work or restrict AI. The problem isn't using AI; it's submitting code you can't understand or defend. AI-generated code carries the same ownership as handwritten code: if you submit it, you own it. Going forward, every bug-fix PR requires a regression test, not just this one. This slows PRs slightly, but builds real judgment and cuts repeat risk.

    464 / 600 characters

  5. 05
    Versioning — I would keep every version of a definition and record which version was used for each submission. When the rules change, the new rules apply to new records only. The 10,000 old records keep the rules they were originally checked against. If old records must follow the new rules, I would update them through a separate, controlled migration. This adds complexity, but keeps the history clear and prevents unexpected changes to old records.

    439 / 600 characters

  6. 06
    Editors — I would let client admins manage their definitions through a guarded UI, long-term. Engineering-managed changes are safer but become a bottleneck as clients grow; direct JSON editing is too easy to get wrong. I'd build a server-side definition linter first, used by engineering in the interim, then reused as the UI's own validation before every published change creates a new version. Cost: more development and maintenance, but safe self-service without depending on engineers for every update.

    496 / 600 characters

Closing Note

Question

What did you leave out, and what are you least confident about? Telling us plainly that you do not know how to do something well is a good answer, not a weak one.

I must admit that the EventFlow is vague and lacks informattion that will have serious implication and business impact on the system design. However, i don't want to make assumptions but rather focus on the core functionalities. I created the "Unknowns" for further clarification if this is in real life development.

I left out nice-to-haves like ticket types (i.e. early birds, regular, vip, student), sale windows, discounts, optional waitlist (i.e. when an event sells out) or even the simple feature of adding the date in the attendees calendar.

I am least confident on the matter of peak demand model as the "2,000 registrations in the first hour after a popular event opens" may mean 1,000 registrations in 30 seconds or 5 minutes. I have not done this in demo or any real deployed systems.
